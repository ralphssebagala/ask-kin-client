const express = require('express');
const router = express.Router();
const db = require('../config/db');
const oracledb = require('oracledb');
const { createPaymentOrder, checkPaymentStatus } = require('../paymentService');

function normalizePhone(phone) {
  if (!phone) return '';
  let p = String(phone).replace(/[\s+\-()]/g, '').replace(/^\+/, '').trim();
  if (p.startsWith('0')) p = p.substring(1);
  if (/^(256|254|255|250|257)\d{8,9}$/.test(p)) return p;
  if (/^7\d{8}$/.test(p)) return '256' + p;
  return p;
}
function isValidPhone(phone) {
  const p = normalizePhone(phone);
  return /^(2567\d{8}|2547\d{8}|2557\d{8}|2507\d{8}|257\d{8,9})$/.test(p);
}

router.post('/init', async (req, res) => {
  const conn = await db.getConnection();
  try {
    const { campaignId, payAmount, netToCampaign, phone, donorName, coverFees, email, donorEmail } = req.body;

    const amountNum = parseFloat(payAmount);
    if (!amountNum || amountNum < 1000) return res.status(400).json({ error: 'Min UGX 1,000' });
    if (!phone ||!isValidPhone(phone)) {
      return res.status(400).json({ error: `Valid number needed. You sent: ${phone}` });
    }

    const cleanPhone = normalizePhone(phone);
    const donorEmailVal = email || donorEmail || null;
    console.log(`Phone fix: ${phone} -> ${cleanPhone} | Email: ${donorEmailVal || 'NULL (guest)'}`);

    const campCheck = await conn.execute(
      `SELECT ID, CAMPAIGN_CURRENCY FROM CAMPAIGNS WHERE ID = :id`,
      [campaignId],
      { outFormat: oracledb.OUT_FORMAT_OBJECT }
    );
    if (campCheck.rows.length === 0) return res.status(404).json({ error: 'Campaign not found' });

    const donationId = Date.now().toString();
    const finalName = donorName && donorName.trim()? donorName.trim() : 'Anonymous';
    const fee = (payAmount - netToCampaign) || 0;

    // FIXED: Column count now matches value count exactly - 12 columns = 12 values
    await conn.execute(
      `INSERT INTO DONATIONS
       (ID, CAMPAIGN_ID, AMOUNT, NET_AMOUNT, FEE, DONOR_NAME, DONOR_PHONE, DONOR_EMAIL, IS_ANONYMOUS, STATUS, ORDER_TRACKING_ID, CREATED_AT)
       VALUES
       (:id, :cid, :amt, :net, :fee, :name, :phone, :email, :anon, :status, :otId, SYSTIMESTAMP)`,
      {
        id: donationId,
        cid: campaignId,
        amt: payAmount,
        net: netToCampaign,
        fee: fee,
        name: finalName,
        phone: cleanPhone,
        email: donorEmailVal, // NULL allowed for guest
        anon: finalName === 'Anonymous'? 1 : 0,
        status: 'PENDING',
        otId: null
      },
      { autoCommit: false }
    );
    await conn.commit();

    const order = await createPaymentOrder({
      amount: payAmount,
      phone: cleanPhone,
      campaignId,
      donationId,
      currency: campCheck.rows[0].CAMPAIGN_CURRENCY || 'UGX'
    });

    await conn.execute(
      `UPDATE DONATIONS SET ORDER_TRACKING_ID = :otId, PESAPAL_TRACKING_ID = :otId WHERE ID = :id`,
      { otId: order.order_tracking_id, id: donationId },
      { autoCommit: true }
    );

    res.json({
      success: true,
      donationId,
      orderTrackingId: order.order_tracking_id,
      redirectUrl: order.redirect_url,
      redirect_url: order.redirect_url
    });

  } catch (err) {
    await conn.rollback().catch(()=>{});
    console.error('Init error:', err.message);
    console.error(err);
    res.status(500).json({ error: 'Failed to init payment', detail: err.message });
  } finally {
    await conn.close().catch(()=>{});
  }
});

router.post('/ipn', async (req, res) => {
  const conn = await db.getConnection();
  try {
    const { OrderTrackingId, OrderMerchantReference } = req.body;
    const status = await checkPaymentStatus(OrderTrackingId);
    if (status.payment_status_description === 'COMPLETED') {
      const donationRes = await conn.execute(
        `SELECT CAMPAIGN_ID, NET_AMOUNT, AMOUNT FROM DONATIONS WHERE ID = :id OR ORDER_TRACKING_ID = :otId`,
        { id: OrderMerchantReference, otId: OrderTrackingId },
        { outFormat: oracledb.OUT_FORMAT_OBJECT }
      );
      if (donationRes.rows.length > 0) {
        const row = donationRes.rows[0];
        await conn.execute(
          `UPDATE DONATIONS SET STATUS = 'COMPLETED' WHERE ID = :id OR ORDER_TRACKING_ID = :otId`,
          { id: OrderMerchantReference, otId: OrderTrackingId },
          { autoCommit: false }
        );
        await conn.execute(
          `UPDATE CAMPAIGNS SET RAISED = NVL(RAISED,0) + :amt WHERE ID = :id`,
          { amt: row.NET_AMOUNT || row.AMOUNT, id: row.CAMPAIGN_ID },
          { autoCommit: false }
        );
        await conn.commit();
      }
    }
    res.json({ status: 'OK' });
  } catch (err) {
    console.error('IPN error:', err);
    res.json({ status: 'FAILED' });
  } finally {
    await conn.close().catch(()=>{});
  }
});

router.get('/:campaignId', async (req, res) => {
  let conn;
  try {
    conn = await db.getConnection();
    const result = await conn.execute(
      `SELECT ID, CAMPAIGN_ID as "campaignId", AMOUNT as "amount", NET_AMOUNT as "netAmount", DONOR_NAME as "donorName", CREATED_AT as "createdAt", STATUS as "status"
       FROM DONATIONS WHERE CAMPAIGN_ID = :id AND STATUS='COMPLETED' ORDER BY CREATED_AT DESC`,
      [req.params.campaignId],
      { outFormat: oracledb.OUT_FORMAT_OBJECT }
    );
    res.json(result.rows);
  } catch (err) {
    res.json([]);
  } finally {
    if(conn) await conn.close().catch(()=>{});
  }
});

router.post('/dev-complete/:id', async (req, res) => {
  if (process.env.NODE_ENV === 'production') {
    return res.status(404).json({ error: 'Not found' });
  }
  let conn;
  try {
    conn = await db.getConnection();
    const { id } = req.params;
    // Search by BOTH donation ID and OrderTrackingId so auto-complete always works
    const dRes = await conn.execute(
      `SELECT ID, CAMPAIGN_ID, NET_AMOUNT, AMOUNT, STATUS FROM DONATIONS WHERE ID = :id OR ORDER_TRACKING_ID = :id`,
      { id: id },
      { outFormat: oracledb.OUT_FORMAT_OBJECT }
    );
    if (dRes.rows.length === 0) {
      console.log("DEV-COMPLETE: Donation not found for", id);
      return res.status(404).json({ error: 'Donation not found: ' + id });
    }
    const donation = dRes.rows[0];
    if (donation.STATUS === 'COMPLETED') {
      return res.json({ success: true, message: 'Already COMPLETED' });
    }
    const amountToAdd = donation.NET_AMOUNT || donation.AMOUNT;
    const realId = donation.ID;

    console.log(`DEV-COMPLETE: Marking ${realId} COMPLETED + adding ${amountToAdd} to campaign ${donation.CAMPAIGN_ID}`);

    await conn.execute(`UPDATE DONATIONS SET STATUS='COMPLETED' WHERE ID = :id`, { id: realId }, { autoCommit: false });
    await conn.execute(
      `UPDATE CAMPAIGNS SET RAISED = NVL(RAISED,0) + :amt, DONORS = NVL(DONORS,0) + 1 WHERE ID = :cid`,
      { amt: amountToAdd, cid: donation.CAMPAIGN_ID },
      { autoCommit: false }
    );
    await conn.commit();
    res.json({ success: true, message: `Donation ${realId} marked COMPLETED`, added: amountToAdd, campaignId: donation.CAMPAIGN_ID });
  } catch (err) {
    if (conn) await conn.rollback().catch(()=>{});
    console.error("DEV-COMPLETE error:", err);
    res.status(500).json({ error: err.message });
  } finally {
    if (conn) await conn.close().catch(()=>{});
  }
});
module.exports = router;
